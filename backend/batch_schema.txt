CREATE TABLE `batches` (
  `id` int NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `course_id` int NOT NULL,
  `instructor_id` int DEFAULT NULL,
  `duration_days` int DEFAULT '30',
  `price` decimal(10,2) DEFAULT '0',
  `enrollment_status` enum('open','closed') DEFAULT 'closed',
  `batch_status` enum('upcoming','active','completed','closed') DEFAULT 'upcoming',
  `enrollment_end_date` datetime DEFAULT NULL,
  `start_date` datetime DEFAULT NULL,
  `end_date` datetime DEFAULT NULL,
  `session_link` varchar(255) DEFAULT NULL,
  `session_time` varchar(255) DEFAULT NULL,
  `session_date` datetime DEFAULT NULL,
  `session_message` text DEFAULT NULL,
  `is_finalized` tinyint(1) DEFAULT '0',
  `material_link` varchar(255) DEFAULT NULL,
  `material_message` text DEFAULT NULL,
  `broadcast_message` text DEFAULT NULL,
  `broadcast_updated_at` datetime DEFAULT NULL,
  `archived_at` datetime DEFAULT NULL,
  `verification_deadline` datetime DEFAULT NULL,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`) /*T![clustered_index] CLUSTERED */,
  KEY `fk_1` (`course_id`),
  KEY `fk_2` (`instructor_id`),
  CONSTRAINT `fk_1` FOREIGN KEY (`course_id`) REFERENCES `courses` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_2` FOREIGN KEY (`instructor_id`) REFERENCES `users` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin AUTO_INCREMENT=30001
